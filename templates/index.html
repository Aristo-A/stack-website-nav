class TwoStacksBrowser:
    def __init__(self):
        self.back_stack = []
        self.forward_stack = []
        self.current_page = None

    def visit(self, url):
        # 1. Push current page to back_stack
        if self.current_page:
            self.back_stack.append(self.current_page)
        # 2. Update current page
        self.current_page = url
        # 3. LIFO RULE: Visiting a new page CLEARS forward history!
        self.forward_stack.clear()

    def back(self):
        if not self.back_stack:
            return False
        # Push current to forward stack
        self.forward_stack.append(self.current_page)
        # Pop previous page from back stack
        self.current_page = self.back_stack.pop()

    def forward(self):
        if not self.forward_stack:
            return False
        self.back_stack.append(self.current_page)
        self.current_page = self.forward_stack.pop()
class DLLNode:
    def __init__(self, url):
        self.url = url
        self.prev = None
        self.next = None

class DoublyLinkedListBrowser:
    def visit(self, url):
        new_node = DLLNode(url)
        if self.current_node:
            # Sever forward chain
            self.current_node.next = new_node
            new_node.prev = self.current_node
        self.current_node = new_node

    def back(self):
        if self.current_node and self.current_node.prev:
            self.current_node = self.current_node.prev

    def forward(self):
        if self.current_node and self.current_node.next:
            self.current_node = self.current_node.next
class ArrayBrowser:
    def __init__(self):
        self.history = []
        self.current_index = -1

    def visit(self, url):
        # Slice array up to current index (discards forward)
        if self.current_index >= 0:
            self.history = self.history[:self.current_index + 1]
        self.history.append(url)
        self.current_index = len(self.history) - 1

    def back(self):
        if self.current_index > 0:
            self.current_index -= 1

    def forward(self):
        if self.current_index < len(self.history) - 1:
            self.current_index += 1
